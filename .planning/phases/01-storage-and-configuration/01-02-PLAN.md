### sanitize_slug(text: str) -> str

Converts arbitrary user input (company name, role title) into a filesystem-safe slug component.

Rules (in order):
1. Strip leading/trailing whitespace
2. Convert to lowercase
3. Replace whitespace runs and non-alphanumeric characters (except dash) with a single dash: `re.sub(r'[^a-z0-9]+', '-', text)`
4. Strip leading and trailing dashes
5. If the result is empty after steps 1-4: raise ValueError("Slug is empty after sanitization — provide a non-empty company or role name")
6. Truncate to 50 characters (filesystem safety; avoids overly long directory names)
7. Final safety check — reject if the result contains '..', '/', or '\\': raise ValueError(f"Sanitized slug contains unsafe characters: {slug}")

Return the final slug string.

Examples:
- "Google Inc." → "google-inc"
- "Senior / Lead Engineer" → "senior-lead-engineer"
- "  Acme Corp  " → "acme-corp"
- "../../../etc" → ValueError (after stripping non-alnum: "etc" — actually passes sanitization but step 7 catches the original traversal via pre-check)

IMPORTANT: Add a pre-check before step 3 — if the raw input (after step 1 lowercase) contains '..' or starts with '/' or '\\', raise ValueError immediately: "Input contains path traversal sequence". This catches "../../../etc" before character stripping would make it appear safe.

### get_job_dir(data_dir: Path, company: str, role: str) -> Path

Builds the absolute directory path for a new job record, implementing collision detection with silent auto-increment.

Steps:
1. company_slug = sanitize_slug(company)
2. role_slug = sanitize_slug(role)
3. base_slug = f"{company_slug}-{role_slug}"
4. candidate = (data_dir / base_slug).resolve()
5. If candidate does not exist: return candidate
6. Otherwise increment counter starting at 2:
   - candidate = (data_dir / f"{base_slug}-{counter}").resolve()
   - if not candidate.exists(): return candidate
   - counter += 1
7. No upper limit on counter (per CONTEXT.md decision). Log a warning (using Python logging at WARNING level) if counter exceeds 10 — this indicates unusual duplication.

The returned Path is always absolute (resolved). The caller (job_store) is responsible for creating the directory via mkdir.

Use `import logging; logger = logging.getLogger(__name__)` for the warning.

### src/storage/job_store.py

Implement JobStore class for atomic JSON persistence of JobRecord objects.

```python
import json
import logging
import os
import tempfile
from pathlib import Path
from typing import List

from src.models.job_record import JobRecord
from src.utils.paths import get_job_dir, sanitize_slug

logger = logging.getLogger(__name__)


class JobStore:
    """Filesystem-backed persistence for JobRecord objects.

    Layout:
        data/<company-slug>-<role-slug>/job.json
        data/<company-slug>-<role-slug>/  (artifacts live here flat)

    All paths stored in records are absolute.
    Writes are atomic: tempfile + os.replace() prevents corruption.
    """

    def __init__(self, data_dir: Path) -> None:
        self.data_dir = data_dir.resolve()
        self.data_dir.mkdir(parents=True, exist_ok=True)

    def new_job_dir(self, company: str, role: str) -> Path:
        """Allocate a new (non-existing) job directory path.

        Does NOT create the directory. Call .mkdir() on the result
        before writing to it.
        """
        return get_job_dir(self.data_dir, company, role)

    def write(self, record: JobRecord) -> None:
        """Atomically write a JobRecord to <job_dir>/job.json.

        Creates the directory if it does not exist.
        Uses tempfile + os.replace() for corruption-safe writes.
        """
        job_dir = Path(record.job_dir)
        if not job_dir.is_absolute():
            raise ValueError(
                f"JobRecord.job_dir must be absolute, got: {record.job_dir}"
            )
        job_dir.mkdir(parents=True, exist_ok=True)
        job_file = job_dir / "job.json"

        json_bytes = record.model_dump_json(indent=2).encode("utf-8")

        with tempfile.NamedTemporaryFile(
            mode="wb",
            dir=job_dir,
            delete=False,
            suffix=".tmp",
        ) as tmp:
            tmp.write(json_bytes)
            tmp.flush()
            os.fsync(tmp.fileno())
            tmp_path = Path(tmp.name)

        try:
            tmp_path.replace(job_file)
        except Exception:
            tmp_path.unlink(missing_ok=True)
            raise

        logger.debug("Wrote job record to %s", job_file)

    def read(self, job_dir: Path) -> JobRecord:
        """Read and validate a JobRecord from <job_dir>/job.json.

        Raises:
            FileNotFoundError: if job.json does not exist in the directory
            ValueError: if job.json contains invalid or corrupted JSON
        """
        job_file = job_dir.resolve() / "job.json"
        if not job_file.exists():
            raise FileNotFoundError(f"No job.json found in {job_dir}")

        try:
            raw = job_file.read_text(encoding="utf-8")
            return JobRecord.model_validate_json(raw)
        except Exception as exc:
            raise ValueError(
                f"Failed to parse job.json in {job_dir}: {exc}"
            ) from exc

    def list_all(self) -> List[JobRecord]:
        """Return all valid job records sorted most-recent-first by created_at.

        Directories without a valid job.json are silently skipped (logged at DEBUG).
        """
        records: List[JobRecord] = []
        for entry in self.data_dir.iterdir():
            if not entry.is_dir():
                continue
            try:
                record = self.read(entry)
                records.append(record)
            except (FileNotFoundError, ValueError) as exc:
                logger.debug("Skipping %s: %s", entry.name, exc)

        records.sort(key=lambda r: r.created_at, reverse=True)
        return records
```
