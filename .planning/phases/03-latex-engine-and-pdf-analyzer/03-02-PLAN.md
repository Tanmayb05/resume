**Function signature and return contracts:**

```python
def compile_tex(tex_path: Path, output_dir: Path) -> dict:
    """Compile a .tex file to PDF via tectonic CLI.

    Args:
        tex_path:   Absolute path to the .tex file to compile.
        output_dir: Directory where tectonic writes the output PDF.
                    Must exist before calling (caller's responsibility).

    Returns:
        On success:
            {"success": True, "pdf_path": Path, "stderr": str | None}
        On failure:
            {"success": False, "error_type": str, "stderr": str, "return_code": int | None}
        error_type values: "compilation_failed" | "timeout" | "tectonic_not_found"
    """
```

**Implementation:**

```python
import subprocess
from pathlib import Path


def compile_tex(tex_path: Path, output_dir: Path) -> dict:
    try:
        result = subprocess.run(
            ["tectonic", "--outdir", str(output_dir), str(tex_path)],
            capture_output=True,
            text=True,
            timeout=60,   # 60s — accounts for first-run package download
            check=True,   # Raises CalledProcessError if return code != 0
        )
        # tectonic names output PDF after the stem of the input .tex file
        pdf_path = output_dir / (tex_path.stem + ".pdf")
        return {
            "success": True,
            "pdf_path": pdf_path,
            "stderr": result.stderr if result.stderr else None,
        }
    except subprocess.CalledProcessError as exc:
        return {
            "success": False,
            "error_type": "compilation_failed",
            "stderr": (exc.stderr[:500] if exc.stderr else "No error output"),
            "return_code": exc.returncode,
        }
    except subprocess.TimeoutExpired:
        return {
            "success": False,
            "error_type": "timeout",
            "stderr": "tectonic compilation exceeded 60 second timeout",
            "return_code": None,
        }
    except FileNotFoundError:
        return {
            "success": False,
            "error_type": "tectonic_not_found",
            "stderr": "tectonic CLI not found in PATH. Install: brew install tectonic",
            "return_code": None,
        }
```

def check_ats_parsability(pdf_path: Path) -> bool:
    """Verify PDF contains extractable text via pdftotext (ATS parsability).

    Returns True if pdftotext extracts non-empty text, False otherwise.
    Returns True if pdftotext is not installed (tool unavailable — assume OK).
    """
    try:
        result = subprocess.run(
            ["pdftotext", str(pdf_path), "-"],
            capture_output=True,
            text=True,
            timeout=10,
            check=False,  # Non-zero return code is not always a failure
        )
        return len(result.stdout.strip()) > 0
    except FileNotFoundError:
        # pdftotext (poppler) not installed — cannot verify, assume OK
        return True
    except subprocess.TimeoutExpired:
        return False
```

def analyze_pdf(pdf_path: Path | str) -> dict:
    """Analyze a PDF for page count and whitespace ratio using PyMuPDF.

    Args:
        pdf_path: Path to the PDF file to analyze.

    Returns:
        On success:
            {
                "success": True,
                "page_count": int,          # Number of pages (>1 = overflow signal)
                "whitespace_ratio": float,   # 0.0 (full text) to 1.0 (all white)
                "page_height_inches": float, # Height of first page in inches
            }
        On error (missing / corrupt / unreadable PDF):
            {
                "success": False,
                "page_count": -1,
                "whitespace_ratio": -1.0,
                "error": str,               # Actionable error message
            }
    """
```

**Implementation — whitespace ratio calculation:**

The whitespace_ratio = 1.0 - (total_text_area / total_page_area).
- total_page_area: sum of (page.rect.width * page.rect.height) across all pages
- total_text_area: sum of bounding box areas from page.get_text("blocks") — each block is (x0, y0, x1, y1, text, ...)
- Guard against zero page_area: `if total_page_area > 0` else whitespace_ratio = 1.0
- Use float arithmetic throughout; clamp result to [0.0, 1.0] with max(0.0, min(1.0, ratio))

**Page height for overflow detection** (Research Pitfall 3 — single tall page):
- `page_height_inches = doc[0].rect.height / 72.0` (PDF points = 1/72 inch)
- If page_height_inches > 11.5 and page_count == 1: this is an overflow signal in Phase 4

```python
import fitz  # PyMuPDF
from pathlib import Path


def analyze_pdf(pdf_path: "Path | str") -> dict:
    try:
        doc = fitz.open(str(pdf_path))
        page_count = doc.page_count

        total_page_area = 0.0
        total_text_area = 0.0

        for page_idx in range(page_count):
            page = doc[page_idx]
            page_area = float(page.rect.width) * float(page.rect.height)
            total_page_area += page_area

            blocks = page.get_text("blocks")
            for block in blocks:
                # block: (x0, y0, x1, y1, "text", block_no, block_type)
                block_w = float(block[2]) - float(block[0])
                block_h = float(block[3]) - float(block[1])
                # Only count positive-area text blocks (type==0 is text)
                if block_w > 0 and block_h > 0:
                    total_text_area += block_w * block_h

        # Page height in inches (for overflow detection in Phase 4)
        first_page_height_in = float(doc[0].rect.height) / 72.0 if page_count > 0 else 0.0

        doc.close()

        if total_page_area > 0:
            text_fill = total_text_area / total_page_area
            whitespace_ratio = max(0.0, min(1.0, 1.0 - text_fill))
        else:
            whitespace_ratio = 1.0

        return {
            "success": True,
            "page_count": page_count,
            "whitespace_ratio": float(whitespace_ratio),
            "page_height_inches": round(first_page_height_in, 4),
        }

    except FileNotFoundError:
        return {
            "success": False,
            "page_count": -1,
            "whitespace_ratio": -1.0,
            "error": f"PDF not found: {pdf_path}",
        }
    except Exception as exc:
        return {
            "success": False,
            "page_count": -1,
            "whitespace_ratio": -1.0,
            "error": f"Error analyzing PDF: {exc}",
        }
```
